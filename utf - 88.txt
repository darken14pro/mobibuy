function escapeHTML(text){

    return String(text)
    .replace(/&/g,"&amp;")
    .replace(/</g,"&lt;")
    .replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;")
    .replace(/'/g,"&#039;");

}

let selectedImage = "";


function previewImage(event){

    const file =
        event.target.files[0];

    if(!file){
        return;
    }

    if(!file.type.startsWith("image/")){

        showToast(
            "Faqat rasm tanlang!"
        );

        return;

    }

    const reader =
        new FileReader();

    reader.onload =
        function(e){

            selectedImage =
                e.target.result;

            const preview =
                document.getElementById(
                    "imagePreview"
                );

            preview.src =
                selectedImage;

            preview.style.display =
                "block";

        };

    reader.readAsDataURL(file);

}

function addProduct(){

    if(!isAdmin){

        showToast(
            "Avval admin sifatida kiring!"
        );

        return;

    }


    const name =
        document
        .getElementById("newName")
        .value

        );

        );

        showToast(
            "Rasm tanlang!"
        );

        return;

    }


    const newProduct = {

        id:Date.now(),

        name:name,

        price:price,

        category:category,

        image:selectedImage

    };


    products.push(newProduct);

    saveProducts();

    renderProducts();

    document
    .getElementById("newPrice")

    .getElementById("newImageFile")

        document.getElementById(

        );

    preview.style.display =
        "none";

    );

/* =====================================================
   EDIT PRICE
===================================================== */

function editPrice(id){

    if(!isAdmin){

        showToast(
            "Faqat admin narxni o‘zgartira oladi!"
        );

        return;

    }


    const product =
        products.find(
            p => p.id === id
        );


    if(!product){
        return;
    }


    const newPrice =
        prompt(
            "Yangi narxni kiriting:",
            product.price
        );


    if(newPrice === null){
        return;
    }


    const number =
        Number(newPrice);


    if(
        isNaN(number) ||
        number < 0
    ){

        showToast(
            "Narx noto‘g‘ri!"
        );

        return;

    }


    product.price =
        number;

    saveProducts();

    renderProducts();

    showToast(
        "Narx o‘zgartirildi ✓"
    );

}


/* =====================================================
   DELETE PRODUCT
===================================================== */

function deleteProduct(id){

    if(!isAdmin){
        return;
    }


    const product =
        products.find(
            p => p.id === id
        );


    if(!product){
        return;
    }


    const answer =
        confirm(
            `"${product.name}" mahsulotini o‘chirasizmi?`
        );


    if(!answer){
        return;
    }


    products =
        products.filter(
            p => p.id !== id
        );


    saveProducts();

    renderProducts();

    showToast(
        "Mahsulot o‘chirildi"
    );

}
